\tikzset{
  font=\sffamily\scriptsize,
  >={Stealth[length=2mm,width=1.5mm]},
  state/.style={draw=#1line, fill=#1fill, rounded corners=5pt, line width=0.8pt,

                font=\sffamily\footnotesize},

% =====================================================================
% 1. Ciclo di vita di un incidente (status su Mongo x claim su Redis)
% =====================================================================
% Versione semplificata: solo eventi e guardie essenziali. Il rinnovo
% idempotente del claim e i dettagli delle scritture stanno nei diagrammi
% di sequenza e nel testo.
\begin{tikzpicture}
\node[init] (i) at (2.0,4.3) {};
\node[state] (open) at (2.0,3.0) {\textbf{Open}};
\node[state] (esc)  at (2.0,0.0) {\textbf{Escalated}};

\node[state, minimum width=2.4cm] (pOpen) at (7.4,3.0) {status = \texttt{open}};
\node[state, minimum width=2.4cm] (pEsc)  at (7.4,0.0) {status = \texttt{escalated}};
\node[ctitle=st, anchor=south west] (ptitle) at ($(pOpen.north west)+(-0.05,0.15)$) {In progress \textmd{(claimed)}};
\begin{scope}[on background layer]
  \node[composite=st, fit=(ptitle)(pOpen)(pEsc)] (prog) {};
\end{scope}

\node[state=ok] (closed) at (12.4,1.5) {\textbf{Closed}};
\node[final] (f) at (14.0,1.5) {};

\draw[tr] (i) -- node[lab, right=2pt] {\ev{create}} (open.north);

% Escalation SLA: anche sui ticket presi in carico.
\draw[tr] (open) -- node[lab, left] {\ev{SLA tick}\\{[}5 min{]}} (esc);
\draw[tr] (pOpen) -- node[lab, right] {\ev{SLA tick}} (pEsc);

\draw[tr] ($(open.east)+(0,0.18)$) -- node[lab, above] {\ev{claim}} ($(pOpen.west)+(0,0.18)$);
\draw[tr] ($(pOpen.west)+(0,-0.18)$) -- node[lab, below, pos=0.6] {\ev{release}, \ev{TTL expiry}} ($(open.east)+(0,-0.18)$);
\draw[tr] ($(esc.east)+(0,0.18)$) -- node[lab, above] {\ev{claim}} ($(pEsc.west)+(0,0.18)$);
\draw[tr] ($(pEsc.west)+(0,-0.18)$) -- node[lab, below, pos=0.6] {\ev{release}, \ev{TTL expiry}} ($(esc.east)+(0,-0.18)$);

\draw[tr] (prog.east |- closed) -- node[lab, above] {\ev{resolve}} node[lab, below] {{[}owner, version{]}} (closed);
\draw[tr] (closed) -- (f);
\end{tikzpicture}

% =====================================================================
% 3. Sessione del client (browser)
% =====================================================================
\begin{tikzpicture}
\node[init] (i) at (0,3.0) {};
\node[state=end, minimum width=2.3cm] (out) at (1.75,3.0) {\textbf{Logged out}};
\node[state, minimum width=2.4cm] (conn) at (5.9,3.0) {\textbf{Connecting}};
\node[state, minimum width=2.4cm] (hyd) at (10.6,3.0) {\textbf{Hydrating}};
\node[state=ok, minimum width=2.4cm] (live) at (10.6,0.3) {\textbf{Live}};

\draw[tr] (i) -- (out);
\draw[tr] ($(out.east)+(0,0.18)$) -- node[lab, above] {\ev{login}} ($(conn.west)+(0,0.18)$);
\draw[tr] ($(conn.west)+(0,-0.18)$) -- node[lab, below] {\ev{unauthorized}} ($(out.east)+(0,-0.18)$);
\draw[tr] (conn) -- node[lab, above] {\ev{connect}\\/ generation++} (hyd);

\draw[tr] ($(hyd.south)+(-0.3,0)$) -- node[lab, left] {\ev{snapshot}\\{[}current{]}} ($(live.north)+(-0.3,0)$);
\draw[tr] ($(live.north)+(0.3,0)$) -- node[lab, right] {\ev{resync\_required}\\/ generation++} ($(hyd.south)+(0.3,0)$);

\draw[tr] ($(hyd.north)+(-0.35,0)$) .. controls +(-0.2,0.8) and +(0.2,0.8) .. ($(hyd.north)+(0.35,0)$);
\node[lab, anchor=south] at ($(hyd.north)+(0,0.62)$) {\ev{snapshot} {[}stale{]} / discard};

\draw[tr] ($(live.east)+(0,0.25)$) .. controls +(0.8,0.2) and +(0.8,-0.2) .. ($(live.east)+(0,-0.25)$);
\node[lab, anchor=west, align=left] at ($(live.east)+(0.75,0)$) {\ev{event} / apply};

\draw[tr] (live.west) -| node[lab, left=2pt, pos=0.75] {\ev{disconnect}} (conn.south);
\draw[tr] (live.south) -- ++(0,-0.5) -| node[lab, above, pos=0.25] {\ev{logout}} (out.south);
\end{tikzpicture}
